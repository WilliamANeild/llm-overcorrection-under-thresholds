format_size() {
    local size=$1
    if [ $size -lt 1024 ]; then
        echo "${size}B"
    elif [ $size -lt 1048576 ]; then
        echo "$((size / 1024))KB"
    elif [ $size -lt 1073741824 ]; then
        echo "$((size / 1048576))MB"
    else
        echo "$((size / 1073741824))GB"
    fi
}

# Check if correct number of arguments provided
if [ $# -ne 2 ]; then
    echo "Error: Incorrect number of arguments."
    usage
fi

SOURCE_DIR="$1"
BACKUP_DEST="$2"

    echo "Error: Source directory '$SOURCE_DIR' does not exist."
    exit 1
fi

# Check if backup destination exists, create if it doesn't

    echo "Backup destination '$BACKUP_DEST' does not exist. Creating it..."
    mkdir -p "$BACKUP_DEST"

        echo "Error: Failed to create backup destination directory."
        exit 1
    fi

# Check if backup was successful

    # Get file size

        echo "Archive: $BACKUP_PATH"

def validate_email(email):
    # Check for no spaces
    if ' ' in email:
        return False  # Email contains a space, so it's invalid
    
    # Check for exactly one @ symbol
    if email.count('@') != 1:
        return False  # Email doesn't have exactly one @ symbol, so it's invalid
    
    # Split the email into local part and domain
    local_part, domain = email.split('@')  # Split at the @ symbol
    
    # Check for something before the @ symbol
    if not local_part:  # Check if local part is empty
        return False  # Email doesn't have a local part, so it's invalid

    if '.' not in domain:  # Check if there's no dot in the domain
        return False  # Email domain doesn't have a dot, so it's invalid
    
    return True  # Email has passed all checks, so it's valid
